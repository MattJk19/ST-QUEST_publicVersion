// 1. Funciones internas o específicas para dibujar tus diagramas
function drawMiDiagrama(data = {}) {
  // Aquí generas la lógica para tu SVG personalizado
  const width = data.width || 400;
  const height = data.height || 300;
  
  // Interacción opcional
  const clickAttrs = data.clickId ? ` data-click-answer="${data.clickId}" class="clickable"` : '';
  
  return `<svg viewBox="0 0 ${width} ${height}" xmlns="http://www.w3.org/2000/svg">
      <rect x="10" y="10" width="100" height="100" fill="blue" ${clickAttrs} />
      <text x="60" y="60" text-anchor="middle" fill="white">${data.texto || 'Hola'}</text>
  </svg>`;
}

// 2. CONTRATO OBLIGATORIO CON EL MOTOR: EL DESPACHADOR
// El motor siempre llamará a esta función pasándole el spec del Question Bank.
function buildDiagramSVG(spec = {}) {
  switch(spec.kind) {
      case 'mi_tipo_diagrama': 
          return drawMiDiagrama(spec.data || {});
      // Agrega más "cases" según los tipos de diagramas de tu materia
      default:
          return "<p>Diagrama no reconocido.</p>";
  }
}
